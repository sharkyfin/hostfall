cmake_minimum_required(VERSION 3.20)
project(HostfallCore VERSION 0.1.0 LANGUAGES CXX)
add_library(hostfall_core STATIC src/graph.cpp)
target_include_directories(hostfall_core PUBLIC include)
target_compile_features(hostfall_core PUBLIC cxx_std_20)
set_target_properties(hostfall_core PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(hostfall_core PRIVATE /W4 /WX)
else()
  target_compile_options(hostfall_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(graph_tests tests/graph_tests.cpp)
  target_link_libraries(graph_tests PRIVATE hostfall_core)
  add_test(NAME graph_invariants COMMAND graph_tests)
endif()
